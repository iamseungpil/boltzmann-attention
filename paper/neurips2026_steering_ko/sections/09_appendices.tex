\section{증명}\label{app:proofs}

\subsection{정리~\ref{thm:beta-star} 완전 증명}\label{app:proof:beta-star}

\paragraph{Setup.}
$Q_\beta := (I + \beta BB^\top) Q$로 $\beta$-섭동된 query를 쓰자. 각 row $q_\beta = (I + \beta BB^\top) q$이고, 로짓은 $\ell_\beta(t) = q_\beta^\top k_t / \sqrt d = \ell_0(t) + \beta r_t$ where $r_t = \tfrac{1}{\sqrt d}\langle BB^\top q, k_t\rangle$. 여기서 $\langle BB^\top q, k_t\rangle = q^\top BB^\top k_t$의 대칭성에서 $r_t = \tfrac{1}{\sqrt d} q^\top BB^\top k_t$로 쓸 수도 있다. $p_\beta(t) = \mathrm{softmax}(\ell_\beta(\cdot))(t)$로 두면 softmax gradient identity는 다음과 같다:
\begin{equation}\label{eq:softmax-grad}
\partial_\beta p_\beta(t) = p_\beta(t) \Big(\partial_\beta \ell_\beta(t) - \mathbb{E}_{p_\beta}[\partial_\beta \ell_\beta]\Big) = p_\beta(t) \big(r_t - \mathbb{E}_{p_\beta}[r]\big).
\end{equation}

\paragraph{1단계: $L(\beta)$의 1차 미분.}
$L(\beta) = \sum_{t\in\mathcal{G}} p_\beta(t)$에 \eqref{eq:softmax-grad}를 적용하고 $\beta=0$에서 평가:
\begin{align*}
\left.\frac{dL}{d\beta}\right|_{\beta=0}
&= \sum_{t\in\mathcal{G}} \partial_\beta p_\beta(t)\big|_{\beta=0}
 = \sum_{t\in\mathcal{G}} p_0(t)(r_t - \bar r) \\
&= \sum_{t\in\mathcal{G}} p_0(t) r_t \;-\; \bar r \sum_{t\in\mathcal{G}} p_0(t).
\end{align*}

\paragraph{2단계: 조건부 평균 표현.}
$\pi_{\mathcal{G}} = \sum_{t\in\mathcal{G}} p_0(t)$, $\bar r_{\mathcal{G}} = \frac{1}{\pi_{\mathcal{G}}} \sum_{t\in\mathcal{G}} p_0(t) r_t$로 두면
\begin{equation*}
\left.\frac{dL}{d\beta}\right|_{\beta=0} = \pi_{\mathcal{G}}\,\bar r_{\mathcal{G}} - \bar r\,\pi_{\mathcal{G}} = \pi_{\mathcal{G}}(\bar r_{\mathcal{G}} - \bar r).
\end{equation*}

\paragraph{3단계: 부호.}
$\pi_{\mathcal{G}} > 0$(ground-truth에 양의 baseline 질량이 걸려 있다는 온화한 조건) 아래에서 $\operatorname{sign}\!\left(\frac{dL}{d\beta}\big|_{\beta=0}\right) = \operatorname{sign}(\bar r_{\mathcal{G}} - \bar r)$. 따라서 원점 근방에서 $L$을 증가시키는 국소 방향은 $\bar r_{\mathcal{G}} - \bar r$의 부호와 일치한다. $\square$

\paragraph{보충: 전역 vs 국소.}
위 논증은 원점에서의 일차 최적성만 확립한다. 전역적으로는 $L(\beta)$의 비선형성과 softmax 포화 때문에 최적 $|\beta|$와 부호가 따로 분석되어야 한다. 본문은 이 정리를 전역 최적화 정리로 쓰지 않고, 표~\ref{tab:sign-routing}의 regime 해석 도구로만 사용한다.

\subsection{정리~\ref{thm:bound} 증명}\label{app:proof:bound}

\emph{1단계.} softmax는 실해석적이므로 $\phi$는 smooth, Taylor의 정리로 $\phi(1)-\phi(0) = \phi'(0) + \int_0^1(1-\tau)\phi''(\tau) d\tau$.
\emph{2단계.} softmax Jacobian $\partial s_t/\partial\ell_{t'} = s_t(\delta_{tt'}-s_{t'})$로 $ds_t(\tau)/d\tau = s_t(\tau)(\alpha_t - \bar\alpha(\tau))$이고 $\phi'(0) = \sum_t s_t\alpha_t(v_t-o) = L(q,E)$ (정확 등식).
\emph{3단계.} 직접 계산으로 $\phi''(\tau) = \sum_t s_t(\tau)[(\alpha_t-\bar\alpha(\tau))^2 - \mathrm{Var}_s\alpha(\tau)] v_t$.
\emph{4단계.} 벡터 Cauchy-Schwarz로 $\|L\|^2 \le \mathrm{Var}_s\alpha(q)\cdot\mathrm{Var}_s[V](q)$, $|\alpha_t|\le Q_{\max}\rho/\sqrt d$로 $\|R\|^2 \le Q_{\max}^4 V_{\max}^2\rho^4/d^2$. $\square$

\subsection{따름정리}\label{app:corollaries}

\begin{corollary}[정지 K의 반복 편향]\label{cor:seka-repeat}
그리디 디코딩 아래 첫 단계에 $t_1\in T_{\mathrm{gt}}$가 선택되고 $t_1$이 온톨로지 부분공간에 정렬되면 정지 K가 $t_1$의 logit을 $(1+\alpha)$배로 증폭해 $\Pr[t_2=t_1\mid\text{stat}] \ge \Pr[t_2=t_1\mid\text{no\_steer}]$. $\alpha=0$일 때 등호.
\end{corollary}

\section{Signed Q 진단의 G-robustness와 fallback 규칙}\label{app:asr-fallback}\label{app:G-robustness}

알고리즘~\ref{alg:asr}은 proxy ground-truth 집합 $\mathcal{G}$ 정의에 의존한다. $\mathcal{G}$를 세 가지 대안으로 정의해 robustness를 측정한다: (i) 도구 이름 단일 토큰, (ii) 도구 이름 전체 토큰 스팬의 평균, (iii) 정답 액션 스트링 전체. 세 정의 모두에서 표~\ref{tab:sign-routing}의 방향 진단이 4개 도메인 중 3개(Airline 제외)에서 유지되면 $\mathcal{G}$ 정의에 대한 민감도가 낮다는 의미다. Airline 경계 사례에서는 $|\bar r_{\mathcal{G}} - \bar r| < \theta$ ($\theta = 0.05$로 설정)일 때 부호를 강제 해석하지 않고 레이어 적응형 K+Q로 fallback하는 규칙을 둔다.
