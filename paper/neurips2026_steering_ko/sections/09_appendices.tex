\section{증명}\label{app:proofs}

\paragraph{정리~\ref{thm:bound} 증명.}\label{app:proof:bound}
\emph{1단계.} softmax는 실해석적이므로 $\phi$는 smooth, Taylor의 정리로 $\phi(1) - \phi(0) = \phi'(0) + \int_0^1(1-\tau)\phi''(\tau)d\tau$.
\emph{2단계.} softmax Jacobian $\partial s_t/\partial\ell_{t'} = s_t(\delta_{tt'}-s_{t'})$로 $ds_t(\tau)/d\tau = s_t(\tau)(\alpha_t - \bar\alpha(\tau))$. 따라서 $\phi'(0) = \sum_t s_t\alpha_t(v_t - o) = L(q,E)$ (정확 등식).
\emph{3단계.} 직접 계산으로 $\phi''(\tau) = \sum_t s_t(\tau)[(\alpha_t - \bar\alpha(\tau))^2 - \mathrm{Var}_s\alpha(\tau)] v_t$.
\emph{4단계.} 벡터 Cauchy-Schwarz로 $\|L\|^2 \le \mathrm{Var}_s\alpha(q)\cdot\mathrm{Var}_s[V](q)$. $|\alpha_t|\le Q_{\max}\rho/\sqrt d$로 $\|R\|^2 \le Q_{\max}^4 V_{\max}^2 \rho^4/d^2$. $\square$

\section{따름정리와 조건부 명제}\label{app:corollaries}

\begin{corollary}[정지 K의 반복 편향]\label{cor:seka-repeat}
그리디 디코딩 아래 첫 단계에 $t_1\in T_{\mathrm{gt}}$가 선택되고 $t_1$이 온톨로지 부분공간에 정렬되면 정지 K가 $t_1$의 logit을 $(1+\alpha)$배로 증폭해 $\Pr[t_2=t_1\mid\text{stat}] \ge \Pr[t_2=t_1\mid\text{no\_steer}]$.
\end{corollary}

\begin{proposition}[레이어 적응형 역할 분리, 조건부]\label{prop:ladapt-intuition}
transformer 표현이 초기 (토큰 인코딩), 중간 (안정화), 후기 (출력 수렴)의 세 단계로 진화한다는 경험적 전제 E1 아래에서, 정지 K는 후기 수렴을 교란하지만 초기 $\lfloor\tau L\rfloor$ 층에 국한된 K는 도구 방향을 각인하여 precision을 높인다.
\end{proposition}

\begin{corollary}[K-imprinting의 시퀀스 길이 감쇠]\label{cor:regime}
명제~\ref{prop:ladapt-intuition}의 전제와 정지성 결합 시, $k$-길이 시퀀스의 $j$번째 도구($j>1$)에 대한 K-imprinting 한계 이득은 $(1-\varphi(j-1))$ 꼴로 상계. Q-커버리지의 per-step 이득은 $j$에 불변.
\end{corollary}
