\paragraph{Proof of Theorem~\ref{thm:bound}.}\label{app:proof:bound}
\emph{Step 1.} Since softmax is real-analytic, $\phi$ is smooth; Taylor's theorem gives $\phi(1) - \phi(0) = \phi'(0) + \int_0^1(1-\tau)\phi''(\tau)d\tau$.
\emph{Step 2.} Softmax Jacobian $\partial s_t/\partial\ell_{t'} = s_t(\delta_{tt'}-s_{t'})$ yields $ds_t(\tau)/d\tau = s_t(\tau)(\alpha_t - \bar\alpha(\tau))$, hence $\phi'(0) = \sum_t s_t \alpha_t(v_t - o) = L(q,E)$ (exact equality).
\emph{Step 3.} A direct calculation gives $\phi''(\tau) = \sum_t s_t(\tau)[(\alpha_t - \bar\alpha(\tau))^2 - \mathrm{Var}_s\alpha(\tau)] v_t$.
\emph{Step 4.} The vector Cauchy--Schwarz yields $\|L\|^2 \le \mathrm{Var}_s\alpha(q)\cdot\mathrm{Var}_s[V](q)$, and $|\alpha_t| \le Q_{\max}\rho/\sqrt d$ gives $\|R\|^2 \le Q_{\max}^4 V_{\max}^2 \rho^4/d^2$. $\square$

\section{Corollaries and Conditional Proposition}\label{app:corollaries}

\begin{corollary}[Repeat bias of stationary K]\label{cor:seka-repeat}
Under greedy decoding, if $t_1 \in T_{\mathrm{gt}}$ is selected at the first step and $t_1$ is aligned with the ontology subspace, then the stationary K amplifies its logit by $(1+\alpha)$, giving $\Pr[t_2 = t_1 \mid \text{stat}] \ge \Pr[t_2 = t_1 \mid \text{no\_steer}]$.
\end{corollary}

\begin{proposition}[Role separation in layer-adaptive, conditional]\label{prop:ladapt-intuition}
Under empirical premise E1 (transformer representations evolve in three stages: early encoding, middle stabilization, late output convergence), applying stationary K across all layers disrupts late-layer convergence, while K-bias restricted to the first $\lfloor \tau L\rfloor$ layers imprints tool direction. Q-coverage is safe across all layers.
\end{proposition}

\begin{corollary}[K-imprinting saturates with sequence length]\label{cor:regime}
Combining E1 with stationarity, the marginal benefit of K-imprinting on the $j$-th tool ($j>1$) decays as $(1-\varphi(j-1))$ for $\varphi\in(0,1)$, while per-step Q-coverage benefit is invariant in $j$.
\end{corollary}
